条件路由允许 Graph 在运行时动态决定执行流向，而不是在定义图结构时静态确定。Nexus 提供两种条件路由机制：ConditionalEdge 和 Command.Goto。

---
路由优先级回顾
Command.Goto（节点内部返回）
    ↓ 无 Command
GraphMiddleware（全局拦截）
    ↓ 无拦截
ConditionalEdge（预定义条件函数）
    ↓ 无条件边
静态 Edge（AddEdge）
    ↓ 无静态边
END（自动结束）

---
ConditionalEdge
基础语法
// Conditional 类型
type Conditional func(ctx context.Context, nexusCtx Context) ([]string, error)
// 添加条件边
g.AddConditionalEdge(fromNode string, condition Conditional)
基本示例：二路分支
g.AddNode("classify", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    // 执行分类逻辑
    sentiment := analyzeSentiment(nctx.Get("text").(string))
    nctx.Set("sentiment", sentiment)
    return nil, nil // 让条件边决定路由
})
g.AddNode("positive_handler", positiveFunc)
g.AddNode("negative_handler", negativeFunc)
g.AddNode("neutral_handler", neutralFunc)
// 添加条件边
g.AddConditionalEdge("classify", func(ctx context.Context, nctx schema.Context) ([]string, error) {
    sentiment := nctx.Get("sentiment").(string)
    switch sentiment {
    case "positive":
        return []string{"positive_handler"}, nil
    case "negative":
        return []string{"negative_handler"}, nil
    default:
        return []string{"neutral_handler"}, nil
    }
})
条件边返回多个节点（并行）
g.AddConditionalEdge("dispatcher", func(ctx context.Context, nctx schema.Context) ([]string, error) {
    // 根据任务类型决定并行执行哪些节点
    taskType := nctx.Get("task_type").(string)
    if taskType == "comprehensive" {
        // 同时执行三个分析
        return []string{"syntax_check", "semantic_check", "style_check"}, nil
    }
    return []string{"quick_check"}, nil
})

Command.Goto 路由
Command.Goto 在节点内部动态决定路由，优先级高于 ConditionalEdge：
g.AddNode("agent_node", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    resp, err := myAgent.Run(ctx, nctx, nil)
    if err != nil {
        return nil, err
    }
    answer := resp.Choices[0].Message.Content
    // 根据 LLM 输出决定路由
    if strings.Contains(answer, "需要人工审核") {
        return schema.GotoNode("human_review"), nil
    }
    if strings.Contains(answer, "任务完成") {
        return schema.GotoEnd(), nil
    }
    // 默认：使用静态边
    return nil, nil
})

多路路由模式
模式 1：基于评分的路由
g.AddConditionalEdge("evaluator", func(ctx context.Context, nctx schema.Context) ([]string, error) {
    score := nctx.Get("quality_score").(float64)
    if score >= 0.9 {
        return []string{"publish"}, nil    // 直接发布
    } else if score >= 0.7 {
        return []string{"review"}, nil     // 人工审核
    } else {
        return []string{"regenerate"}, nil // 重新生成
    }
})
模式 2：动态路由到任意节点（策略模式）
// 在节点中动态决定下一步
g.AddNode("strategy_node", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    // 从 Context 或 LLM 输出获取下一步
    nextNode := nctx.Get("next_agent").(string)  // e.g. "agent_a" 或 "agent_b"
    return schema.GotoNode(nextNode), nil
})
模式 3：循环路由
g.AddNode("retry_node", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    retries := 0
    if v := nctx.Get("retries"); v != nil {
        retries = v.(int)
    }
    retries++
    nctx.Set("retries", retries)
    result := tryOperation()
    if result.OK || retries >= 3 {
        nctx.Set("final_result", result)
        return schema.GotoNode("output"), nil
    }
    // 返回到自身重试
    return schema.GotoNode("retry_node"), nil
})
g.SetEntryPoint("retry_node")
模式 4：基于 LLM 输出的路由
g.AddNode("router_agent", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    // 让 LLM 决定路由
    routerAgent, _ := agent.NewBaseAgent(
        agent.WithName("router"),
        agent.WithDesc("路由决策 Agent"),
        agent.WithLLM(llmClient),
        agent.WithInstruction(`根据用户查询，决定交给哪个专家处理。
            可选选项：["math_expert", "code_expert", "writing_expert"]
            请直接返回专家名称，不要有其他内容。`),
    )
    resp, err := routerAgent.Run(ctx, nctx, nil)
    if err != nil {
        return nil, err
    }
    nextNode := strings.TrimSpace(resp.Choices[0].Message.Content)
    return schema.GotoNode(nextNode), nil
})

ConditionalEdge vs Command 选择指南
场景
推荐方式
基于预先定义的固定条件
ConditionalEdge
节点执行后才能知道路由
Command.Goto
需要 LLM 参与路由决策
Command.Goto（在节点内调用 LLM）
基于共享状态的路由
ConditionalEdge（读取 SharedState）
动态路由到未知节点
Command.Goto

---
错误处理
g.AddConditionalEdge("check", func(ctx context.Context, nctx schema.Context) ([]string, error) {
    value, ok := nctx.GetShared("result")
    if !ok {
        // 返回 error，会导致 Invoke 失败
        return nil, errors.New("result not found in context")
    }
    // 类型安全转换
    score, ok := value.(float64)
    if !ok {
        return nil, fmt.Errorf("invalid score type: %T", value)
    }
    if score > 0.8 {
        return []string{"pass"}, nil
    }
    return []string{"fail"}, nil
})

---
注意事项
1. 每个节点只能有一条条件边：AddConditionalEdge 会覆盖同一节点的上一个条件边。
2. 条件边优先于静态边：如果一个节点同时有条件边和静态边，条件边优先（除非条件函数返回空列表）。
3. 空列表 = 回退到静态边：如果 Conditional 返回 ([]string{}, nil)，图会回退到静态边。
4. 并发条件：条件边的判断函数在主 goroutine 中执行，不会并发调用。
5. 避免循环路由：使用 Command.Goto 或条件边创建循环时，确保有退出条件，否则会无限执行。
