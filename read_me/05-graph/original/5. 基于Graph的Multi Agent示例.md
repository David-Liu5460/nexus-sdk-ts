Nexus 的图编排天然支持多 Agent 协作，不同 Agent 作为独立节点，通过共享 Context 传递状态。

---
多 Agent 协作模式
模式 1：顺序协作（流水线）
每个 Agent 依次执行，上一个 Agent 的输出作为下一个 Agent 的输入：
// 三个 Agent 的流水线：分析 → 规划 → 执行
package main
import (
    "context"
    "os"
    "code.byted.org/ad/nexus/agent"
    "code.byted.org/ad/nexus/callback"
    "code.byted.org/ad/nexus/context/oceanaicontext"
    "code.byted.org/ad/nexus/llm"
    "code.byted.org/ad/nexus/llm/openai"
    "code.byted.org/ad/nexus/schema"
    goopenai "code.byted.org/ad/go-openai"
)
func main() {
    ctx := context.Background()
    llmClient, _ := openai.New(
        openai.WithToken(os.Getenv("OPENAI_API_KEY")),
        openai.WithModel("gpt-4o"),
        openai.WithBaseURL(os.Getenv("OPENAI_BASE_URL")),
        openai.WithAPIType(goopenai.APITypeAzure),
    )
    nctx, _ := oceanaicontext.NewOceanAIContext(
        oceanaicontext.WithAppName("my-app"),
        oceanaicontext.WithUserName("user"),
        oceanaicontext.WithReq(&schema.AgentRunRequest{
            Query: "帮我制定一个学习 Go 语言的计划",
        }),
    )
    eventCB := callback.NewEventCallback(
        callback.WithCustomHandler(func(ctx context.Context, nctx schema.Context, event *schema.Event) {
            // 处理事件
        }),
    )
    g := schema.NewGraph(nctx, schema.WithCallback(eventCB))
    // Agent 1：需求分析
    g.AddNode("analyzer", func(ctx context.Context, nctx schema.Context,
        reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
        a, _ := agent.NewBaseAgent(
            agent.WithName("analyzer"),
            agent.WithDesc("需求分析 Agent"),
            agent.WithLLM(llmClient),
            agent.WithInstruction("分析用户需求，提取关键信息。"),
            agent.WithCallback(cb),
        )
        resp, err := a.Run(ctx, nctx, nil)
        if err != nil {
            return nil, err
        }
        nctx.Set("analysis", resp.Choices[0].Message.Content)
        return nil, nil
    })
    // Agent 2：计划制定
    g.AddNode("planner", func(ctx context.Context, nctx schema.Context,
        reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
        analysis := nctx.Get("analysis").(string)
        p, _ := agent.NewBaseAgent(
            agent.WithName("planner"),
            agent.WithDesc("计划制定 Agent"),
            agent.WithLLM(llmClient),
            agent.WithInstruction("基于需求分析，制定具体的执行计划。"),
            agent.WithCallback(cb),
        )
        resp, err := p.Run(ctx, nctx,
            []schema.ChatCompletionMessage{
                llm.NewUserMessage("planner", "需求分析结果："+analysis+"\n请制定执行计划。"),
            },
        )
        if err != nil {
            return nil, err
        }
        nctx.Set("plan", resp.Choices[0].Message.Content)
        return nil, nil
    })
    // Agent 3：执行总结
    g.AddNode("executor", func(ctx context.Context, nctx schema.Context,
        reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
        plan := nctx.Get("plan").(string)
        e, _ := agent.NewBaseAgent(
            agent.WithName("executor"),
            agent.WithDesc("执行总结 Agent"),
            agent.WithLLM(llmClient),
            agent.WithInstruction("将计划转化为可执行的步骤清单。"),
            agent.WithCallback(cb),
        )
        resp, err := e.Run(ctx, nctx,
            []schema.ChatCompletionMessage{
                llm.NewUserMessage("executor", "执行计划："+plan+"\n请生成详细步骤。"),
            },
        )
        if err != nil {
            return nil, err
        }
        nctx.Set("final_output", resp.Choices[0].Message.Content)
        return nil, nil
    })
    g.AddEdge("analyzer", "planner")
    g.AddEdge("planner", "executor")
    g.AddEdge("executor", schema.END)
    g.SetEntryPoint("analyzer")
    app, _ := g.Compile()
    app.Invoke(ctx, schema.ChatCompletionMessage{
        Role:    schema.ChatMessageRoleUser,
        Content: "帮我制定一个学习 Go 语言的计划",
    })
}

模式 3：监督者-工作者（Supervisor-Worker）
Supervisor Agent 决定调用哪个 Worker Agent：
g.AddNode("supervisor", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    supervisor, _ := agent.NewBaseAgent(
        agent.WithName("supervisor"),
        agent.WithDesc("任务分派主管"),
        agent.WithLLM(llmClient),
        agent.WithInstruction(`你是一个任务分派主管。
根据用户请求，决定分配给哪个专家：
- code_expert: 编程相关问题
- math_expert: 数学计算问题
- writing_expert: 文字创作问题
请直接输出专家名称。`),
        agent.WithCallback(cb),
    )
    resp, err := supervisor.Run(ctx, nctx, nil)
    if err != nil {
        return nil, err
    }
    nextAgent := strings.TrimSpace(resp.Choices[0].Message.Content)
    return schema.GotoNode(nextAgent), nil
})
// 各专家节点
g.AddNode("code_expert", codeExpertFunc)
g.AddNode("math_expert", mathExpertFunc)
g.AddNode("writing_expert", writingExpertFunc)
// 汇聚到输出
g.AddEdge("code_expert", "output")
g.AddEdge("math_expert", "output")
g.AddEdge("writing_expert", "output")
g.AddEdge("output", schema.END)
g.SetEntryPoint("supervisor")

多 Agent 共享 Context
共享状态（Shared State）
所有 Agent 可读写，适合传递中间结果：
// Agent A 写入
nctx.SetShared("analysis_result", analysisData)
// Agent B 读取
result, ok := nctx.GetShared("analysis_result")
私有状态（Private State）
每个 Agent 独立的命名空间，避免并行冲突：
// Agent A 写入私有状态
nctx.SetPrivate("agent_a", "my_result", resultA)
// 其他 Agent 不会看到 agent_a 的私有状态
// 但 Agent A 可以读取自己的私有状态
result, _ := nctx.GetPrivate("agent_a", "my_result")

---
多 Agent 消息隔离
每个 BaseAgent 实例有唯一的 namespace（短 hash），消息按命名空间存储，互不干扰：
// Agent A 的消息存储在 "abc123" 命名空间
agentA.Run(ctx, nctx, nil)  // 内部：nctx.AddMessage("abc123", ...)
// Agent B 的消息存储在 "def456" 命名空间
agentB.Run(ctx, nctx, nil)  // 内部：nctx.AddMessage("def456", ...)
// 通过 Roadmap 整合所有历史
roadmap := nctx.LoadRoadmap()  // 包含所有 Agent 的摘要

---
注意事项
1. 并行节点的竞态：并行执行时（GotoNodes 或多条 AddEdge），多个 Agent 同时写入相同 SharedState key 会产生竞态条件。使用 SetPrivate 隔离写入。
2. Agent 名称唯一：不同节点中创建的 Agent 如果使用相同名称，会共享消息命名空间（如果 namespace 相同），通常这是预期行为（多轮对话中同一 Agent 的不同调用）。
3. Callback 传递：建议将 Graph 的 callback 参数传递给节点内创建的 Agent，以确保事件流完整。
4. 避免深度递归：图的执行是递归的，深度嵌套的 Graph（一个 Graph 节点中运行另一个 Graph）可能导致 goroutine 栈溢出。
