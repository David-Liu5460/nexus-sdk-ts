自定义 Agent
当内置的 BaseAgent、SkillAgent 等无法满足需求时，可以通过实现 schema.Agent 接口来创建完全自定义的 Agent。

> 📘 TypeScript 版本见 [`3.自定义Agent (TS).md`](../3.自定义Agent%20(TS).md)，基于 Bun + TS SDK 实际代码编写，含 Go ↔ TS API 映射表与差异标注。

---
schema.Agent 接口

```go
type Agent interface {
    Run(ctx context.Context, nexusCtx Context,
        messages []ChatCompletionMessage,
        opts ...GenerateOption) (*ChatCompletionResponse, error)
    Name() string
    Description() string
    Stop()
}
```

---
最简实现示例

```go
package myagent
import (
    "context"
    "code.byted.org/ad/nexus/schema"
)
// 编译时接口检查
var _ schema.Agent = (*EchoAgent)(nil)
// EchoAgent：将用户最后一条消息原样返回
type EchoAgent struct {
    name string
    stop bool
}
func NewEchoAgent(name string) *EchoAgent {
    return &EchoAgent{name: name}
}
func (a *EchoAgent) Name() string        { return a.name }
func (a *EchoAgent) Description() string { return "将用户输入原样返回的 Echo Agent" }
func (a *EchoAgent) Stop()               { a.stop = true }
func (a *EchoAgent) Run(
    ctx context.Context,
    nctx schema.Context,
    messages []schema.ChatCompletionMessage,
    opts ...schema.GenerateOption,
) (*schema.ChatCompletionResponse, error) {
    if a.stop {
        return nil, schema.ErrStop
    }
    // 获取用户最后一条消息
    var lastContent string
    if len(messages) > 0 {
        lastContent = messages[len(messages)-1].Content
    } else {
        history := nctx.LoadAllMessages(a.name, "")
        if len(history) > 0 {
            lastContent = history[len(history)-1].Content
        }
    }
    // 构造响应
    return &schema.ChatCompletionResponse{
        Choices: []schema.ChatCompletionChoice{
            {
                Message: schema.ChatCompletionMessage{
                    Role:    schema.ChatMessageRoleAssistant,
                    Content: "Echo: " + lastContent,
                    Name:    a.name,
                },
                FinishReason: schema.FinishReasonStop,
            },
        },
    }, nil
}
```

带 LLM 的自定义 Agent
以下是一个包含自定义逻辑的 Agent 示例：

```go
package myagent
import (
    "context"
    "fmt"
    "code.byted.org/ad/nexus/llm"
    "code.byted.org/ad/nexus/schema"
)
type ChainOfThoughtAgent struct {
    name   string
    llm    schema.LLM
    stop   bool
}
func NewChainOfThoughtAgent(name string, llm schema.LLM) *ChainOfThoughtAgent {
    return &ChainOfThoughtAgent{name: name, llm: llm}
}
func (a *ChainOfThoughtAgent) Name() string        { return a.name }
func (a *ChainOfThoughtAgent) Description() string { return "使用思维链的推理 Agent" }
func (a *ChainOfThoughtAgent) Stop()               { a.stop = true }
func (a *ChainOfThoughtAgent) Run(
    ctx context.Context,
    nctx schema.Context,
    messages []schema.ChatCompletionMessage,
    opts ...schema.GenerateOption,
) (*schema.ChatCompletionResponse, error) {
    if a.stop {
        return nil, schema.ErrStop
    }
    // Step 1: 先让 LLM 产生思维链
    cotPrompt := []schema.ChatCompletionMessage{
        llm.NewSystemMessage(a.name, "请先逐步思考，再给出最终答案。格式：\n思考：...\n答案：..."),
    }
    if len(messages) > 0 {
        cotPrompt = append(cotPrompt, messages...)
    } else {
        cotPrompt = append(cotPrompt, nctx.LoadAllMessages(a.name, "")...)
    }
    // Step 2: 调用 LLM
    resp, err := a.llm.GenerateContent(ctx, nctx, cotPrompt, opts...)
    if err != nil {
        return nil, fmt.Errorf("LLM call failed: %w", err)
    }
    // Step 3: 保存结果到 Context
    if len(resp.Choices) > 0 {
        nctx.AddMessage(a.name, resp.Choices[0].Message)
    }
    return resp, nil
}
```

在 Graph 中使用自定义 Agent

```go
g.AddNode("custom_node", func(ctx context.Context, nctx schema.Context,
    reg func(func(schema.Context)), cb schema.Callback) (*schema.Command, error) {
    myAgent := myagent.NewChainOfThoughtAgent("cot_agent", llmClient)
    // 注册停止函数
    reg(func(nctx schema.Context) {
        myAgent.Stop()
    })
    resp, err := myAgent.Run(ctx, nctx, nil)
    if err != nil {
        return nil, err
    }
    nctx.Set("cot_result", resp.Choices[0].Message.Content)
    return nil, nil
})
```

---
最佳实践
1. 编译时接口检查

```go
// 在包级别添加，确保接口实现完整
var _ schema.Agent = (*MyAgent)(nil)
```

2. 正确处理 Stop 信号

```go
func (a *MyAgent) Run(...) (*schema.ChatCompletionResponse, error) {
    // 在循环开始时检查
    for i := 0; i < maxIter; i++ {
        if a.stop {
            return nil, schema.ErrStop
        }
        // ...
    }
}
```

3. 遵循消息命名空间约定

```go
// 使用 Agent 名称作为消息命名空间
nctx.AddMessage(a.name, messages...)
contextMessages := nctx.LoadAllMessages(a.name, a.namespace)
```

4. 触发自动压缩
如果你的 Agent 有多轮循环，建议在完成后触发压缩：

```go
defer func() {
    _ = nctx.AutoCompaction(a.name, a.namespace)
}()
```

5. 支持 Callback

```go
type MyAgent struct {
    callback schema.Callback
}
func (a *MyAgent) Run(ctx context.Context, nctx schema.Context,
    messages []schema.ChatCompletionMessage, opts ...schema.GenerateOption) (*schema.ChatCompletionResponse, error) {
    if a.callback != nil {
        a.callback.AgentStartCallback(ctx, nctx, a, messages, nil)
        defer func() {
            a.callback.AgentEndCallback(ctx, nctx, a, resp, err, nil)
        }()
    }
    // ...
}
```
